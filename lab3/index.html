โค้ดโครงร่าง (Skeleton Code)

ศึกษาโค้ดต่อไปนี้และค้นหาคำตอบเพื่อนำมาเติมในช่องว่างทางขวามือให้สมบูรณ์:

webserver_littlefs.ino 
#if defined(ESP8266)
#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
ESP8266WebServer server(80);
#elif defined(ESP32)
#include <WiFi.h>
#include <WebServer.h>
WebServer server(80);
#endif
#include <LittleFS.h>

const char* ssid = "iot_512";
const char* password = "iot123456";

void handleFileRequest() {
  String path = server.uri();
  if (path.endsWith("/")) path += "index.html";
  
  String dataType = "text/plain";
  if (path.endsWith(".html")) dataType = "text/html";
  else if (path.endsWith(".css")) dataType = "text/css";
  else if (path.endsWith(".js")) dataType = "application/javascript";
  
  if (LittleFS.exists(path)) {
    File file = LittleFS.open(path, "r");
    server.streamFile(file, dataType);
    file.close();
  } else {
    server.send(404, "text/plain", "File Not Found");
  }
}

void setup() {
  Serial.begin(115200);
  
  // 1. เชื่อมต่อ Wi-Fi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println(WiFi.localIP());
  
  // 2. เริ่มต้นเมานต์ระบบไฟล์ LittleFS
  #if defined(ESP8266)
  if(!LittleFS.begin()){
  #else
  if(!LittleFS.begin(true)){
  #endif
     Serial.println("LittleFS Mount Failed");
     return;
  }
  
  // 3. ตั้งเส้นทางเซิร์ฟเวอร์ให้ส่งไฟล์แบบ Dynamic เมื่อมีการเรียกหน้านอกเหนือจากที่ระบุ
  server.onNotFound(  );
  server.begin();
  Serial.println("HTTP Web Server started.");
}

void loop() {
  // 4. จัดการประมวลผลคำสั่งฝั่งไคลเอนต์ที่ส่งเข้ามาที่เว็บเซิร์ฟเวอร์
  server.;
}

ทฤษฎีที่เกี่ยวข้อง: เว็บเซิร์ฟเวอร์บนบอร์ด ESP32 / ESP8266 (LittleFS Web Server)

1.1 การทำงานของ Embedded Web Server

ไมโครคอนโทรลเลอร์ทำหน้าที่เป็นเซิร์ฟเวอร์เปิดรับการเชื่อมต่อ TCP พอร์ต 80 ตามมาตรฐาน HTTP:

  • HTTP Request: บราวเซอร์ส่งคำขอไฟล์ เช่น GET /index.html HTTP/1.1
  • HTTP Response: บอร์ดส่ง Header รหัสสถานะ (เช่น 200 OK หรือ 404 Not Found) พร้อมส่งข้อมูล Content-Type และตัวไฟล์กลับไป

2.1 ระบบแฟ้มข้อมูล LittleFS Flash Memory

LittleFS ถูกออกแบบมาเฉพาะสำหรับ Flash Memory ที่มีทรัพยากรจำกัดในอุปกรณ์ IoT:

  • Power-loss Resilient: ป้องกันระบบไฟล์เสียหายเมื่อไฟดับกะทันหัน
  • Wear Leveling: กระจายการเขียนข้อมูลไปยังบล็อกต่างๆ ของ Flash ยืดอายุการใช้งานฮาร์ดแวร์
  • จัดเก็บไฟล์เว็บแยกส่วน: ช่วยให้สามารถแยกโค้ด C++ ออกจากไฟล์เว็บ (HTML, CSS, JS) ในโฟลเดอร์ data/ ได้อย่างเป็นระเบียบ

3.1 ประสิทธิภาพของคำสั่ง server.streamFile()

การใช้ server.streamFile(file, contentType) จะอ่านไฟล์จาก Flash แล้วส่งออกทางเครือข่ายทีละก้อน (Chunked Transfer) โดยไม่ต้องโหลดไฟล์ทั้งก้อนเข้ามาเก็บในตัวแปร String บน RAM ช่วยป้องกันปัญหา Heap Overflow (แรมหมด) ทำให้เซิร์ฟเวอร์ตอบสนองได้รวดเร็วและเสถียร
