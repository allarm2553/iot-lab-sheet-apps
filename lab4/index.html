วัตถุประสงค์

  1. เพื่อทำความเข้าใจเทคนิคส่งข้อมูลจากเซ็นเซอร์บนบอร์ดขึ้นสู่เว็บเบราว์เซอร์แบบเรียลไทม์โดยไม่ต้องรีเฟรชหน้า

โค้ดโครงร่าง (ฝั่ง ESP32 / ESP8266 WebSocket Server)

esp_websocket.ino 
#if defined(ESP8266)
#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#elif defined(ESP32)
#include <WiFi.h>
#include <WebServer.h>
#endif
#include <WebSocketsServer.h>
#include <ArduinoJson.h>
#include <LittleFS.h>

const char* ssid = "iot_512";
const char* password = "iot123456";

#if defined(ESP8266)
ESP8266WebServer server(80);
#else
WebServer server(80);
#endif
WebSocketsServer webSocket = WebSocketsServer(81);

void handleFileRequest() {
  String path = server.uri();
  if (path.endsWith("/")) path += "index.html";
  String dataType = "text/plain";
  if (path.endsWith(".html")) dataType = "text/html";
  else if (path.endsWith(".css")) dataType = "text/css";
  else if (path.endsWith(".js")) dataType = "application/javascript";
  
  if (LittleFS.exists(path)) {
    File file = LittleFS.open(path, "r");
    server.streamFile(file, dataType);
    file.close();
  } else {
    server.send(404, "text/plain", "File Not Found");
  }
}

void webSocketEvent(uint8_t num, WStype_t type, uint8_t * payload, size_t length) {
  if (type == WStype_TEXT) {
    JsonDocument doc;
    deserializeJson(doc, payload, length);
    
    // 1. ตรวจสอบและรับสถานะคำสั่งควบคุมรีเลย์จากปุ่มบนหน้าเว็บ
    if(doc.containsKey("action")) {
       String action = doc["action"];
       if(action == "toggle_fan") {
          bool val = doc["value"];
          digitalWrite(13, val ? HIGH : LOW);
       }
    }
  }
}

void broadcastSensorData(float temp, float hum) {
  JsonDocument doc;
  doc["temp"] = temp;
  doc["humidity"] = hum;
  
  String output;
  serializeJson(doc, output);
  
  // 2. ส่งข้อมูลกระจายไปยังหน้าเว็บทุกหน้าจอที่มีการเชื่อมต่อท่อ Socket อยู่
  webSocket.(output);
}

void setup() {
  Serial.begin(115200);
  
  // เชื่อมต่อ Wi-Fi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected.");
  
  // เริ่มต้นเมานต์ระบบไฟล์ LittleFS
  #if defined(ESP8266)
  if(!LittleFS.begin()){
  #else
  if(!LittleFS.begin(true)){
  #endif
     Serial.println("LittleFS Mount Failed");
     return;
  }
  
  // ตั้งค่าเซิร์ฟเวอร์
  server.onNotFound(handleFileRequest);
  server.begin();
  Serial.println("HTTP Web Server started.");
  
  // ตั้งค่า WebSocket
  webSocket.begin();
  webSocket.onEvent(webSocketEvent);
  Serial.println("WebSocket Server started.");
}

void loop() {
  server.handleClient();
  webSocket.loop();
}

ทฤษฎีที่เกี่ยวข้อง: การส่งข้อมูลสองทางแบบเรียลไทม์ (Local WebSockets & JSON)

1.1 ทำไมต้องใช้ WebSockets ในงานควบคุมเรียลไทม์?

HTTP Polling: บราวเซอร์ต้องส่ง Request พร้อม Header ขนาดใหญ่ซ้ำๆ ทุก 1-2 วินาที สิ้นเปลืองแบนด์วิดท์และเกิดความหน่วงสูง (Latency)

WebSockets (Port 81): สร้างการเชื่อมต่อแบบ Full-Duplex Persistent TCP Connection ครั้งเดียว ทำให้ทั้งบอร์ดและบราวเซอร์สามารถส่งข้อมูลสวนกันได้ทันทีเมื่อมีอีเวนต์เกิดขึ้น (Event-driven) ช่วยให้การกดสวิตช์เปิด-ปิดพัดลมและการอ่านค่าเซ็นเซอร์ตอบสนองแบบเสี้ยววินาที

2.1 การเข้ารหัสและถอดรหัส JSON Payload

การใช้ไลบรารี ArduinoJson ช่วยรวมข้อมูลเซ็นเซอร์หลายค่าไว้ในก้อนเดียวกัน:

// การสร้าง JSON ส่งออก (Serialization):
JsonDocument doc;
doc["temp"] = temperature;
doc["fan"] = fanState;
String msg;
serializeJson(doc, msg);
webSocket.broadcastTXT(msg); // ส่งหาทุกบราวเซอร์พร้อมกัน
