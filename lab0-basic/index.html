โค้ดแม่แบบภาษา C++ (Skeleton Code)

src/main.cpp 
#include <Arduino.h>

// กำหนดพินควบคุม LED และปุ่มกด
#define ONBOARD_LED_PIN   2   // LED บนบอร์ด ESP32 (GPIO 2)
#define EXT_LED1_PIN      23  // LED ภายนอกดวงที่ 1 (GPIO 23)
#define EXT_LED2_PIN      19  // LED ภายนอกดวงที่ 2 (GPIO 19)
#define BUTTON_PIN        18  // สวิตช์ปุ่มกด (GPIO 18)

void setup() {
  // เริ่มต้นการสื่อสาร Serial ที่ความเร็ว 115200 Baud
  Serial.begin(115200);
  delay(500);

  // [ช่องว่าง 1]: กำหนดทิศทางการทำงานของพิน LED (INPUT / OUTPUT)
  pinMode(ONBOARD_LED_PIN, );
  pinMode(EXT_LED1_PIN, OUTPUT);
  pinMode(EXT_LED2_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  Serial.println("ESP32 Ready: Initialized GPIO Pins!");
}

void loop() {
  // [ช่องว่าง 2]: สั่งจ่ายระดับแรงดันลอจิกให้หลอด LED (HIGH / LOW)
  digitalWrite(ONBOARD_LED_PIN, );
  digitalWrite(EXT_LED1_PIN, HIGH);
  digitalWrite(EXT_LED2_PIN, LOW);
  delay(500);

  // สั่งให้ LED ดับ (จ่ายสัญญาณลอจิก LOW 0V)
  digitalWrite(ONBOARD_LED_PIN, LOW);
  digitalWrite(EXT_LED1_PIN, LOW);
  digitalWrite(EXT_LED2_PIN, HIGH);
  delay(500);

  Serial.println("Blink Cycle Completed");
}

โจทย์ท้าทายการทดลอง (Challenge: Non-blocking Multi-LED & Strobe)

โจทย์: พัฒนาโปรแกรมไฟกระพริบหลายดวงแบบไม่หยุดค้าง (Non-blocking Blink Controller):

  • 1. การสลับจังหวะไฟกระพริบ 3 สเต็ป: ให้ Onboard LED (GPIO 2), LED สีแดง (GPIO 23) และ LED สีน้ำเงิน (GPIO 19) กระพริบสลับจังหวะ (Alternating / Traffic Strobe)
  • 2. การควบคุมแบบ Non-blocking ด้วย millis(): เปลี่ยนจากการใช้คำสั่ง delay() มาใช้ตัวแปรจับเวลา millis() เพื่อให้ CPU สามารถประมวลผลงานอื่นพร้อมกันได้โดยไม่หยุดค้าง
  • 3. การรับอินพุตปุ่มกด SW1 (GPIO 18): เมื่อกดปุ่ม SW1 ค้างไว้ ให้ไฟกระพริบเร็วขึ้น 4 เท่า (125 ms) และส่งข้อความ Log แจ้งเตือนสถานะผ่าน Serial Monitor (115200 Baud)

1. สถาปัตยกรรมและโครงสร้างพินเอาต์ (ESP32 & ESP8266 Pinout Architecture)

ไมโครคอนโทรลเลอร์ ESP32 และ ESP8266 เป็นชิปประมวลผล System-on-Chip (SoC) ที่ได้รับความนิยมสูงสุดในงาน Internet of Things (IoT) และระบบอัตโนมัติ:

  • ESP32 Microcontroller: ใช้ซีพียู 32-bit Dual-Core 240MHz, มี Wi-Fi 2.4GHz และ Bluetooth 4.2/BLE ในตัว, SRAM ขนาดใหญ่ 520KB และ ADC ความละเอียด 12-bit (0-4095)
  • ESP8266 Microcontroller: ใช้ซีพียู 32-bit Single-Core 80/160MHz, มี Wi-Fi ในตัว, SRAM ขนาด 80KB และ ADC ความละเอียด 10-bit (0-1023) ที่ขา A0 เพียงขาเดียว
แผนผังพินบอร์ด ESP32 Dev Module (30-Pin NodeMCU ESP32 / IPST-WiFi)
Xtensa Dual-Core 240MHz
แผนผังพินบอร์ด ESP8266 NodeMCU V2 / V3 (ESP-12E / ESP-12F Module)
Tensilica L106 80/160MHz
คุณสมบัติ (Features) ESP32 (ESP-WROOM-32) ESP8266 (NodeMCU ESP-12E/F)
หน่วยประมวลผล (CPU) Xtensa 32-bit LX6 Dual-Core (240 MHz) Tensilica L106 32-bit Single-Core (80/160 MHz)
หน่วยความจำภายใน (SRAM) 520 KB SRAM 80 KB SRAM (User App ~36 KB)
หน่วยความจำ Flash (SPI) 4 MB (ขยายได้สูงสุด 16 MB) 4 MB
การเชื่อมต่อไร้สาย Wi-Fi 802.11 b/g/n + Bluetooth 4.2 BR/EDR/BLE Wi-Fi 802.11 b/g/n (ไม่มี Bluetooth)
จำนวนขา GPIO ใช้งานได้ ~28-36 ขา (มี Input-Only 4 ขา) ~11 ขา (แชร์กับ Flash และ Strapping)

    • พิมพ์ค้นหา esp8266 โดย ESP8266 Community แล้วกด Install (สำหรับบอร์ด NodeMCU / ESP8266)
  4. ติดตั้ง USB Driver (หากคอมพิวเตอร์มองไม่เห็นพอร์ต COM):
    • สำหรับชิป CP2102 / CP2104: ติดตั้งไดรเวอร์ Silicon Labs CP210x USB to UART Bridge
    • สำหรับชิป CH340 / CH341: ติดตั้งไดรเวอร์ WCH CH340 USB Driver
    • สำหรับชิป CH9102 / FT232: ติดตั้งไดรเวอร์เฉพาะตามชิป USB-to-UART ของบอร์ด
  5. เลือกบอร์ดและพอร์ต:
    • สำหรับ ESP32: ไปที่ Tools -> Board -> esp32 -> ESP32 Dev Module (หรือ IPST-WiFi)
    • สำหรับ ESP8266: ไปที่ Tools -> Board -> esp8266 -> NodeMCU 1.0 (ESP-12E Module)
    • เลือก Tools -> Port ให้ตรงกับพอร์ต USB ของบอร์ด (เช่น COM3, COM5)

3. การติดตั้งและพัฒนาด้วย VS Code, PlatformIO IDE และ Wokwi Simulator

การพัฒนา IoT ระดับมืออาชีพนิยมใช้ VS Code ร่วมกับ PlatformIO เนื่องจากมีระบบจัดการ Library อัตโนมัติและระบบ IntelliSense ที่รวดเร็ว:

  1. ติดตั้ง VS Code: ดาวน์โหลดไฟล์ติดตั้งจาก code.visualstudio.com และติดตั้งตามขั้นตอนมาตรฐาน
  2. ติดตั้งส่วนขยาย PlatformIO IDE: ในหน้าต่าง Extensions (Ctrl+Shift+X) พิมพ์ค้นหา PlatformIO IDE แล้วกด Install (ระบบจะดาวน์โหลด Python และ PIO Core ให้อัตโนมัติ)
  3. ติดตั้งส่วนขยาย Wokwi Simulator: ค้นหา Wokwi Simulator ใน Extensions แล้วกด Install สำหรับจำลองวงจรและทดสอบโค้ดได้โดยไม่ต้องต่อฮาร์ดแวร์จริง
  4. โครงสร้างไฟล์ PlatformIO:
    • platformio.ini: กำหนดประเภทบอร์ด, Framework (Arduino), ความเร็ว Serial Monitor (115200)
    • src/main.cpp: ไฟล์เขียนโค้ดภาษา C++ หลักของระบบ
    • wokwi.toml & diagram.json: ไฟล์กำหนดค่าการเชื่อมโยงวงจรจำลองใน Wokwi

4. คุณลักษณะพิน GPIO, ขา Input-Only และ Strapping Pins

ไมโครคอนโทรลเลอร์ ESP32 มีข้อกำหนดและข้อจำกัดของพินที่ต้องระมัดระวังในการออกแบบวงจร ดังนี้:

  • Input-Only Pins (GPIO 34, 35, 36/VP, 39/VN): เป็นขาที่สามารถรับสัญญาณอินพุต (Digital/Analog) ได้เท่านั้น ไม่สามารถสั่งเป็น OUTPUT หรือจ่ายกระแสไฟได้ และไม่มีวงจร Internal Pull-up/Pull-down ในตัวชิป
  • Strapping Pins (GPIO 0, 2, 5, 12, 15): เป็นขาที่ชิป ESP32 ตรวจสอบระดับแรงดันขณะจ่ายไฟ (Power-on Reset) เพื่อเลือกโหมดการทำงาน:
    • GPIO 0: ต้องเป็น HIGH ตอนรันปกติ (หากเป็น LOW จะเข้าสู่ UART Download Mode)
    • GPIO 2: ต้องเป็น LOW หรือ Floating ขณะดาวน์โหลดเฟิร์มแวร์
    • ข้อควรระวัง: ห้ามต่ออุปกรณ์ภายนอกที่ดึงแรงดันค้างไว้ที่ขาเหล่านี้ เพราะอาจทำให้บอร์ดบูตไม่ขึ้น

5. โครงสร้างโปรแกรม Arduino C++ และการเขียนโค้ด Non-blocking

การเขียนโปรแกรมควบคุมไมโครคอนโทรลเลอร์ ESP32 แบ่งโครงสร้างหลักออกเป็น 2 ส่วน:

  • ฟังก์ชัน setup(): ทำงานเพียงรอบเดียวเมื่อเปิดเครื่อง ใช้สำหรับกำหนดค่าเริ่มต้น เช่น pinMode(pin, OUTPUT) หรือ Serial.begin(115200)
  • ฟังก์ชัน loop(): ทำงานวนซ้ำตลอดเวลา ใช้สำหรับประมวลผลคำสั่ง ควบคุมเอาต์พุต และอ่านค่าเซ็นเซอร์
  • ข้อแตกต่างระหว่าง delay() กับ millis():
    • delay(ms): เป็นแบบ Blocking คือสั่งหยุดการทำงานของ CPU ทั้งหมด ทำให้ไม่สามารถอ่านเซ็นเซอร์หรือสื่อสาร Wi-Fi ได้ระหว่างหน่วงเวลา
    • millis(): คืนค่าเวลาเป็นมิลลิวินาทีตั้งแต่เริ่มเปิดเครื่อง ใช้สร้างตัวจับเวลาแบบ Non-blocking ทำให้ CPU สามารถทำงานอื่นควบคู่ไปได้โดยระบบไม่หยุดค้าง
