วัตถุประสงค์ (Objectives)

  1. เข้าใจโปรโตคอลการสื่อสารแบบอนุกรม I2C (Inter-Integrated Circuit) ซึ่งใช้สายสัญญาณเพียง 2 เส้น ได้แก่ SDA (Data) และ SCL (Clock) พร้อมระบบระบุตัวตนด้วย I2C Address (0x3C)
  2. เข้าใจบทบาทของ Frame Buffer (1,024 ไบต์ใน RAM) และคำสั่ง display.display() ในการส่งข้อมูลขึ้นหน้าจอแบบ Flicker-Free
  3. สามารถใช้งานฟังก์ชันกราฟิกจากไลบรารี Adafruit_GFX (วาดแถบ Header แบบ Inverted Text, วาดเส้นแบ่ง, และสร้าง Progress Bar)
  4. สามารถออกแบบและจัดวาง UI บนหน้าจอขนาด 128x64 พิกเซล ร่วมกับการใช้ Non-blocking Timer (millis()) เพื่อรีเฟรชหน้าจออย่างมีประสิทธิภาพ

โค้ดโครงร่าง (Skeleton Code)

ศึกษาโค้ดต่อไปนี้และค้นหาคำตอบเพื่อนำมาเติมในช่องว่างทางขวามือให้สมบูรณ์:

oled_demo.ino 
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22); // SDA = GPIO 21, SCL = GPIO 22 สำหรับ ESP32
  
  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) { 
    Serial.println(F("SSD1306 allocation failed"));
    for(;;);
  }
  
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
}

void loop() {
  display.clearDisplay();
  
  // 1. เขียนข้อความพาดหัว (Header) ไว้ที่แถวบนสุด
  display.setCursor(0, 0);
  display.print("MONITORING NODE");
  
  // 2. วาดเส้นแบ่งกึ่งกลางหน้าจอแนวนอนที่พิกเซล Y = 10
  display.drawLine(0, 10, 128, 10, );
  
  // 3. แสดงค่าเซ็นเซอร์อุณหภูมิที่อ่านได้ (สมมติค่าอุณหภูมิ = 28.5)
  display.setCursor(0, 15);
  display.printf("Temp: %.1f C", 28.5);
  
  // 4. แสดงผลข้อความส่งคำสั่งออกจอภาพจริง
  display.;
  delay(1000);
}

ทฤษฎีที่เกี่ยวข้อง: การแสดงผลผ่านจอ OLED SSD1306 (I2C Bus & Graphics)

1.1 โปรโตคอลการสื่อสารแบบ I2C (Inter-Integrated Circuit)

I2C เป็นระบบบัสสื่อสารแบบอนุกรมความเร็วสูง (Serial Synchronous Bus) ที่พัฒนาโดย Philips ซึ่งใช้สายสัญญาณเพียง 2 เส้นในการเชื่อมต่ออุปกรณ์ได้หลายสิบตัวพร้อมกัน:

  • SDA (Serial Data): สายรับส่งข้อมูลแบบสองทิศทาง (Bidirectional) ระหว่าง Master (ESP32) และ Slave (OLED SSD1306)
  • SCL (Serial Clock): สายส่งสัญญาณนาฬิกากำหนดจังหวะความเร็วในการส่งข้อมูล (Standard Mode 100 kHz, Fast Mode 400 kHz)
  • I2C 7-bit Address: อุปกรณ์ทุกตัวบนบัสต้องมีที่อยู่ระบุตัวตนไม่ซ้ำกัน โดยจอ OLED SSD1306 นิยมใช้ 0x3C (หรือ 0x3D)

1.2 การกำหนดขาพินฮาร์ดแวร์ I2C บน ESP32 และ ESP8266

ไมโครคอนโทรลเลอร์แต่ละรุ่นมีขาฮาร์ดแวร์ I2C เริ่มต้นแตกต่างกัน:

// สำหรับ ESP32 (IPST-WiFi / NodeMCU-32S)
Wire.begin(21, 22); // SDA = GPIO 21, SCL = GPIO 22

// สำหรับ ESP8266 (AX-WiFi / NodeMCU v2)
Wire.begin(4, 5); // SDA = GPIO 4 (D2), SCL = GPIO 5 (D1)

2.1 สถาปัตยกรรม Frame Buffer ในหน่วยความจำ RAM

จอแสดงผล OLED ชนิด Monochrome ขนาด 128×64 พิกเซล (1 บิตต่อพิกเซล: 0=มืด, 1=สว่าง) ทำงานร่วมกับหน่วยความจำบัฟเฟอร์ในแรมของไมโครคอนโทรลเลอร์ขนาด:

Frame Buffer Size = (128 × 64 พิกเซล) ÷ 8 บิต/ไบต์ = 1,024 ไบต์ (1 KB)

ทุกคำสั่งวาดภาพ เช่น display.print(), display.drawRect(), หรือ display.drawLine() จะแก้ไขบิตแมพใน RAM ชั่วคราวก่อน โดยยังไม่มีการส่งข้อมูลออกทางสาย I2C ทำให้วาดภาพได้รวดเร็วและไม่กระตุก

2.2 บทบาทของคำสั่ง display.display()

เมื่อเตรียมภาพและข้อความใน Frame Buffer เสร็จเรียบร้อย จึงเรียกใช้คำสั่ง display.display() เพื่อสั่งให้ชิปส่งข้อมูลทั้งก้อน 1,024 ไบต์ผ่านบัส I2C ไปยังชิปไดรเวอร์ SSD1306 ในคราวเดียว ช่วยให้หน้าจอไม่เกิดการกะพริบ (Flicker-Free Rendering)

3.1 ฟังก์ชันและคำสั่งพื้นฐานใน Adafruit_GFX

display.clearDisplay();              // ล้างพิกเซลทั้งหมดในบัฟเฟอร์ให้เป็น 0
display.setTextSize(1);                // ขนาดฟอนต์มาตรฐาน (1=6x8 px, 2=12x16 px)
display.setTextColor(SSD1306_WHITE);   // สีตัวอักษรเป็นสีขาว (พิกเซลสว่าง)
display.setCursor(0, 0);               // กำหนดจุดเริ่มต้นเขียนข้อความ (X=0..127, Y=0..63)
display.printf("Temp: %.1f C", temp);  // พิมพ์ข้อความพร้อมฟอร์แมตทศนิยม
display.drawLine(0, 10, 128, 10, SSD1306_WHITE); // วาดเส้นตรงแนวนอน

3.2 ระบบพิกัดหน้าจอ (128x64 Coordinate Grid)

พิกัดเริ่มต้นมุมบนซ้ายสุดคือ (0, 0) โดยแกน X มีค่าตั้งแต่ 0 ถึง 127 (แนวนอน) และแกน Y มีค่าตั้งแต่ 0 ถึง 63 (แนวตั้ง)

4.1 เทคนิคการออกแบบ UI สไตล์ Inverted Header & Gauge

เพื่อให้อุปกรณ์ IoT ดูสวยงามและอ่านข้อมูลง่าย สามารถใช้เทคนิคข้อความกลับสี (Inverted Text) และหลอดเกจแสดงผลแบบกราฟิก:

// วาดแถบหัวข้อสีขาวพาดเต็มขอบบน (Inverted Title Bar)
display.fillRect(0, 0, 128, 10, SSD1306_WHITE);
display.setTextColor(SSD1306_BLACK, SSD1306_WHITE);
display.setCursor(14, 1); display.print("IoT SMART DASHBOARD");

// วาดหลอด Progress Bar แสดงระดับแรงดัน ADC 0-100%
display.drawRect(55, 27, 72, 8, SSD1306_WHITE); // กรอบนอก
display.fillRect(57, 29, barWidth, 4, SSD1306_WHITE); // แถบระดับพลังงาน

5.1 ตารางการตรวจวินิจฉัยและแก้ปัญหา (Troubleshooting Matrix)

อาการที่พบ (Symptom) สาเหตุที่แท้จริง (Root Cause) แนวทางแก้ไข (Solution)
SSD1306 allocation failed สาย SDA/SCL หลวม หรือระบุ I2C Address ผิด (เช่น 0x3D) ตรวจสอบการต่อสายพิน I2C ให้ตรง และเปลี่ยนแอดเดรสเป็น 0x3C
โค้ดรันปกติแต่จอดำสนิท ลืมเรียกคำสั่ง display.display(); หรือลืมจ่ายไฟเลี้ยง VCC 3.3V เพิ่มคำสั่ง display.display(); ท้ายบล็อกการวาดภาพเสมอ
ตัวอักษรและภาพซ้อนทับกันเละ ไม่ได้ล้าง Frame Buffer ก่อนวาดภาพเฟรมใหม่ เรียก display.clearDisplay(); ทุกครั้งก่อนเริ่มวาดข้อมูลใหม่
ระบบหน่วง ปุ่มกดตอบสนองช้า สั่ง display.display() ทุกรอบลูป ทำให้ I2C ติดคอขวด ใช้ Non-blocking Timer millis() รีเฟรชจอที่ความถี่ 2–5 Hz (200–500ms)

6.1 คำสั่งควบคุมหน้าจอ OLED และตั้งค่าผ่าน Message Box

นักศึกษาสามารถพิมพ์คำสั่งผ่าน Serial Monitor Input Box (Baud rate 115200) เพื่อทดสอบการเรนเดอร์หน้าจอ OLED และการจำลองสัญญาณอินพุต:

วัตถุประสงค์ คำสั่งที่ต้องพิมพ์ใน Message Box ผลลัพธ์บนจอ OLED และฮาร์ดแวร์
พิมพ์ข้อความขึ้นจอ OLED MSG=Hello_IoT แสดงข้อความ "Hello_IoT" ในบรรทัดสถานะบนจอ OLED
สั่งเปิด/ปิดพัดลมระบายความร้อน FAN=1 หรือ FAN=0 สลับรีเลย์ GPIO 5 และอัปเดตสถานะ FAN บนจอ OLED ทันที
จำลองค่าระดับความชื้นดิน SOIL=85.5 อัปเดต Gauge และกราฟแท่ง Progress bar บนหน้าจอ
ล้างหน้าจอ OLED ทั้งหมด CLEAR_OLED สั่ง display.clearDisplay(); display.display();
// ตัวอย่างการรับข้อความมาแสดงผลบนจอ OLED:
if (Serial.available()) {

  input.trim();
  if (input.startsWith("MSG=")) {
    customMsg = input.substring(4);
