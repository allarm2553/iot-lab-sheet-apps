วัตถุประสงค์

  1. เพื่อเรียนรู้การเชื่อมต่อจอภาพ OLED ขนาดเล็กผ่านโปรโตคอลการสื่อสารแบบ I2C

โค้ดโครงร่าง (Skeleton Code)

ศึกษาโค้ดต่อไปนี้และค้นหาคำตอบเพื่อนำมาเติมในช่องว่างทางขวามือให้สมบูรณ์:

oled_demo.ino 
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

void setup() {
  Serial.begin(115200);
  #if defined(ESP8266)
  Wire.begin(4, 5); // SDA = GPIO4 (D2), SCL = GPIO5 (D1) สำหรับ ESP8266
  #else
  Wire.begin(21, 22); // SDA = GPIO21, SCL = GPIO22 สำหรับ ESP32
  #endif
  
  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) { 
    Serial.println(F("SSD1306 allocation failed"));
    for(;;);
  }
  
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
}

void loop() {
  display.clearDisplay();
  
  // 1. เขียนข้อความพาดหัว (Header) ไว้ที่แถวบนสุด
  display.setCursor(0, 0);
  display.print("MONITORING NODE");
  
  // 2. วาดเส้นแบ่งกึ่งกลางหน้าจอแนวนอนที่พิกเซล Y = 10
  display.drawLine(0, 10, 128, 10, );
  
  // 3. แสดงค่าเซ็นเซอร์อุณหภูมิที่อ่านได้ (สมมติค่าอุณหภูมิ = 28.5)
  display.setCursor(0, 15);
  display.printf("Temp: %.1f C", 28.5);
  
  // 4. แสดงผลข้อความส่งคำสั่งออกจอภาพจริง
  display.;
  delay(1000);
}

ทฤษฎีที่เกี่ยวข้อง: การแสดงผลผ่านจอ OLED SSD1306 (I2C Bus & Graphics)

1.1 โปรโตคอลการสื่อสารแบบ I2C (Inter-Integrated Circuit)

I2C เป็นระบบบัสสื่อสารแบบอนุกรมความเร็วสูงที่ใช้สายสัญญาณเพียง 2 เส้นในการเชื่อมต่ออุปกรณ์ได้หลายตัวพร้อมกัน:

  • SDA (Serial Data): สายรับส่งบิตข้อมูลระหว่าง Master (ESP) และ Slave (OLED)
  • SCL (Serial Clock): สายสัญญาณนาฬิกากำหนดจังหวะการรับส่งข้อมูล
  • Device Address: อุปกรณ์แต่ละตัวระบุตัวตนด้วยแอดเดรส เช่น จอ OLED SSD1306 ใช้ 0x3C

2.1 สถาปัตยกรรม Frame Buffer ในหน่วยความจำ RAM

จอภาพ OLED ขนาด 128×64 พิกเซล แบบ Monochrome (1 บิตต่อพิกเซล) ใช้บัฟเฟอร์ในแรมขนาด: (128 × 64) / 8 = 1,024 ไบต์ (1 KB)

คำสั่งวาดภาพ เช่น display.print() หรือ display.drawLine() จะเขียนข้อมูลลงบัฟเฟอร์ในแรมก่อน และเมื่อเรียกใช้คำสั่ง display.display() ข้อมูลทั้งเฟรมจะถูกส่งผ่านบัส I2C ขึ้นสู่หน้าจอจริงในครั้งเดียว ช่วยป้องกันอาการจอกะพริบ

3.1 คำสั่งสำคัญในไลบรารี Adafruit_GFX

display.clearDisplay();            // ล้างบัฟเฟอร์จอภาพ
display.setTextSize(1);              // กำหนดขนาดตัวอักษร (1=6x8 px, 2=12x16 px)
display.setTextColor(SSD1306_WHITE); // กำหนดสีตัวอักษรเป็นสีขาว
display.setCursor(0, 10);            // กำหนดพิกัดเริ่มต้น (X=0, Y=10)
display.print("Temp: "); display.print(temp);
display.display();                   // สั่งเรนเดอร์ภาพขึ้นจอจริง
