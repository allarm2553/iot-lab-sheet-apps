โค้ดโครงร่าง (Skeleton Code)

ศึกษาโค้ดต่อไปนี้และค้นหาคำตอบเพื่อนำมาเติมในช่องว่างทางขวามือให้สมบูรณ์:

lab1_skeleton.ino 
#include <Arduino.h>
#include <DHT.h>

#define DHTPIN 33 // (หรือ D3/GPIO 0 สำหรับบอร์ด AX-WiFi)
#define DHTTYPE DHT11
#define ANALOG_PIN 36 // (GPIO 36/KNOB-S สำหรับ IPST-WiFi, หรือ A0 สำหรับบอร์ด AX-WiFi)
#define FAN_RELAY_PIN 5 // (หรือ D7/GPIO 13 สำหรับบอร์ด AX-WiFi)
#define MIST_RELAY_PIN 23 // (หรือ D0/GPIO 16 สำหรับบอร์ด AX-WiFi)

DHT dht(DHTPIN, DHTTYPE);

float temperature = 0;
int rawAnalog = 0;
float analogPercent = 0;

void setup() {
  Serial.begin(115200);
  dht.begin();
  
  // 1. ตั้งค่าโหมดของขารีเลย์ให้เป็นเอาต์พุต
  pinMode(FAN_RELAY_PIN, );
  pinMode(MIST_RELAY_PIN, );
  
  // 2. ตั้งค่าลดทอนสัญญาณของ ADC เป็น 11dB (ช่วงแรงดัน 0-3.3V)
  #if defined(ESP32)
  analogSetPinAttenuation(ANALOG_PIN, ADC_11db);
  #endif
}

void loop() {
  temperature = dht.readTemperature();
  rawAnalog = analogRead(ANALOG_PIN);
  
  // 3. แปลงค่า rawAnalog (0-4095 สำหรับ ESP32 หรือ 0-1023 สำหรับ ESP8266) ให้เป็นเปอร์เซ็นต์ (0-100%)
  analogPercent = (rawAnalog / ) * 100.0;
  
  Serial.printf("Temp: %.1f C | Analog: %.1f%%\n", temperature, analogPercent);
  
  // 4. เขียนเงื่อนไข: หากอุณหภูมิ > 30 องศาเซลเซียส ให้เปิดพัดลม
  if (temperature > 30.0) {
    digitalWrite(FAN_RELAY_PIN, );
  } else {
    digitalWrite(FAN_RELAY_PIN, );
  }
  
  delay(2000);
}

โจทย์ท้าทายการทดลอง (Grand Challenge with Fail-Safe)

โจทย์: พัฒนาระบบควบคุมสภาพแวดล้อมอัจฉริยะแบบทนทานต่อข้อผิดพลาด (Fault-Tolerant & Fail-Safe Smart Climate Controller):

  • 1. ระบบควบคุมพัดลมแบบ Hysteresis: สั่งเปิดพัดลมเมื่ออุณหภูมิ ≥ 30.0 °C และปิดเมื่ออุณหภูมิ ≤ 29.5 °C เพื่อป้องกันรีเลย์กระตุก (Relay Chatter)
  • 2. ระบบควบคุมความชื้น: สั่งเปิดปั๊มพ่นหมอกเมื่อความชื้น ≤ 50% และปิดเมื่อความชื้น ≥ 60%
  • 3. ระบบ Fail-Safe Protection (ความปลอดภัย): ใช้คำสั่ง isnan() ตรวจสอบค่า หากอ่านเซ็นเซอร์ผิดพลาดต่อเนื่อง ให้สั่งตัดการทำงานของพัดลมและปั๊มหมอกทันที (Safe State) พร้อมแจ้งเตือนผ่าน Serial Monitor

เนื้อหาทฤษฎีที่เกี่ยวข้อง: การอ่านค่าเซ็นเซอร์และควบคุมเอาต์พุต

1.1 การกำหนดโหมดพินและการส่งสัญญาณดิจิทัล (Digital I/O)

ไมโครคอนโทรลเลอร์ ESP32/ESP8266 มีขา General Purpose Input/Output (GPIO) ที่สามารถกำหนดหน้าที่ผ่านคำสั่งภาษา C/C++:

  • pinMode(pin, OUTPUT); : กำหนดให้พินทำหน้าที่ส่งแรงดันลอจิกออกภายนอก
  • digitalWrite(pin, HIGH); : จ่ายแรงดันไฟฟ้าระดับลอจิกสูง (3.3V)
  • digitalWrite(pin, LOW); : จ่ายแรงดันไฟฟ้าระดับลอจิกต่ำ (0V / Ground)

1.2 เหตุผลที่ต้องใช้โมดูลรีเลย์ (Relay Module) ขับโหลด

ขา GPIO ของไมโครคอนโทรลเลอร์สามารถจ่ายกระแสไฟฟ้าได้สูงสุดเพียงประมาณ 12 – 20 mA ที่แรงดัน 3.3V ซึ่งไม่สามารถจ่ายพลังงานให้แก่อุปกรณ์กำลังสูง เช่น มอเตอร์พัดลมระบายความร้อน หรือปั๊มน้ำ (ซึ่งต้องการกระแส 100 mA – 1 A+ ที่แรงดัน 5V/12V/220V) ได้โดยตรง

โมดูลรีเลย์ (Relay) ทำหน้าที่เป็นสวิตช์แม่เหล็กไฟฟ้าที่มีวงจร Optocoupler แยกสัญญาณควบคุมฝั่งแรงดันต่ำ (3.3V) ออกจากวงจรขับโหลดกำลังสูง ป้องกันสัญญาณรบกวนและแรงดันไฟเหนี่ยวนำย้อนกลับ (Back EMF) ไม่ให้ทำลายชิปไมโครคอนโทรลเลอร์

2.1 หลักการแปลงสัญญาณแอนะล็อกเป็นดิจิทัล (ADC)

วงจร ADC (Analog-to-Digital Converter) ทำหน้าที่สุ่มวัดระดับแรงดันไฟฟ้าแบบต่อเนื่อง (0 – 3.3V) แล้วแปลงเป็นตัวเลขจำนวนเต็มดิจิทัล (Discrete Digital Value)

คุณสมบัติ ESP32 (IPST-WiFi) ESP8266 (AX-WiFi)
ความละเอียด (Resolution) 12-bit (212 = 4096 ระดับ) 10-bit (210 = 1024 ระดับ)
ช่วงค่าดิบที่อ่านได้ (Raw Value) 0 ถึง 4095 0 ถึง 1023
พินรับสัญญาณแอนะล็อก GPIO 32–36, 39 (พิน 36 คือ KNOB-S) พิน A0 / ADC0 (ปรับจัมเปอร์ไปที่ VR)

2.2 สูตรการแปลงค่าดิบเป็นเปอร์เซ็นต์ (0 - 100%)

เพื่อนำค่าแรงดันไปแสดงผลและประมวลผลต่อในรูปแบบสากล เราจะแปลงค่าดิบจาก ADC ให้เป็นสเกล 0–100%:

// สำหรับ ESP32 (12-bit ADC):
float analogPercent = (rawAnalog / 4095.0) * 100.0;

// สำหรับ ESP8266 (10-bit ADC):
float analogPercent = (rawAnalog / 1023.0) * 100.0;

3.1 โครงสร้างและหลักการทำงานของเซ็นเซอร์ DHT11

DHT11 เป็นโมดูลเซ็นเซอร์วัดสภาพแวดล้อมแบบดิจิทัลที่รวมเซ็นเซอร์ 2 ชนิดไว้ในตัวเดียว:

  • วัดอุณหภูมิ: ใช้ตัวต้านทานแปรค่าตามอุณหภูมิชนิด NTC Thermistor (ความต้านทานลดลงเมื่ออุณหภูมิสูงขึ้น)
  • วัดความชื้นสัมพัทธ์: ใช้แผ่นโพลีเมอร์ดูดซับความชื้นแบบ Resistive Humidity Sensing Component

3.2 การสื่อสารข้อมูลแบบ Single-Bus Protocol

DHT11 ส่งข้อมูลดิจิทัลผ่านสายสัญญาณ DATA เพียงเส้นเดียว โดยในการอ่านค่า 1 ครั้ง เซ็นเซอร์จะส่งข้อมูลมาทั้งหมด 40 บิต (5 ไบต์):

[ไบต์ 1: ค่าจำนวนเต็มความชื้น] + [ไบต์ 2: ค่าทศนิยมความชื้น] +
[ไบต์ 3: ค่าจำนวนเต็มอุณหภูมิ] + [ไบต์ 4: ค่าทศนิยมอุณหภูมิ] +
[ไบต์ 5: Parity Checksum ตรวจสอบความถูกต้อง]

ช่วงการวัด: อุณหภูมิ 0 ถึง 50°C (±2°C) | ความชื้น 20 ถึง 90% RH (±5% RH)

4.1 ปัญหารีเลย์ตัด-ต่อถี่เกินไป (Relay Chatter / Clicking)

หากเราเขียนเงื่อนไขควบคุมแบบจุดตัดเดียว เช่น if (temp > 30.0) ON; else OFF; เมื่ออุณหภูมิจริงอยู่ที่ประมาณ 29.99°C – 30.01°C สัญญาณรบกวนเพียงเล็กน้อยจะทำให้รีเลย์เปิดและปิดสลับไปมาอย่างรวดเร็วหลายครั้งต่อวินาที ส่งผลให้หน้าสัมผัสของรีเลย์สึกหรออย่างรวดเร็วและมอเตอร์พัดลมอาจเสียหาย

4.2 การแก้ไขด้วยเทคนิค Hysteresis (Deadband Control)

การควบคุมแบบ Hysteresis จะกำหนดจุดตัด 2 ระดับเพื่อสร้างช่วงปลอดภัย (Deadband):

  • Upper Threshold (จุดเปิดพัดลม): สั่งเปิดเมื่อ อุณหภูมิ > 30.0°C
  • Lower Threshold (จุดปิดพัดลม): สั่งปิดเมื่อ อุณหภูมิ < 29.5°C
  • ช่วง Hysteresis Band (29.5°C ถึง 30.0°C): รักษาสถานะเดิมไว้ ไม่สั่งเปลี่ยน
if (temperature > 30.0) {
  fanState = true;  // สั่งเปิดพัดลม
} else if (temperature < 29.5) {
  fanState = false; // สั่งปิดพัดลม
}
digitalWrite(FAN_RELAY_PIN, fanState ? HIGH : LOW);

5.1 การดักจับความผิดพลาดของเซ็นเซอร์ด้วย isnan() และ Fail-Safe Action

เมื่อสายสัญญาณเซ็นเซอร์ DHT11 หลุด หลวม หรืออ่านค่าถี่เกิน 2 วินาที ฟังก์ชัน dht.readTemperature() จะคืนค่าเป็น NaN (Not a Number) หากเรานำค่า NaN ไปเปรียบเทียบใน if (temperature > 30) จะได้ผลลัพธ์เป็น false เสมอ ทำให้ระบบไม่สั่งเปิดพัดลมแม้เครื่องจักรจะร้อนจัด!

แนวทางแก้ไข (Safe State): ต้องตรวจสอบด้วยฟังก์ชัน isnan() หากอ่านค่าไม่ได้ ให้ตัดการทำงานของโหลดกำลังสูง (Relay) ทันทีและส่งข้อความเตือน:

if (isnan(temperature) || isnan(humidity)) {
  Serial.println("[ERROR] ไม่สามารถอ่านค่า DHT11 ได้! ตัดการทำงานของโหลดเพื่อความปลอดภัย");
  digitalWrite(FAN_RELAY_PIN, LOW);  // เข้าสู่ Fail-Safe State
  digitalWrite(MIST_RELAY_PIN, LOW);
  return;  // ข้ามลูปการประมวลผลรอบนี้
}

5.2 ปัญหา Integer Division ในการคำนวณ ADC

ในภาษา C/C++ หากนำจำนวนเต็ม (int) มาหารด้วยจำนวนเต็ม เช่น (rawAnalog / 4095) * 100 คอมไพเลอร์จะปัดเศษทศนิยมทิ้งทันที ทำให้ได้ผลลัพธ์เป็น 0 เสมอ (ยกเว้นเมื่อ rawAnalog == 4095 ที่ได้ 1)

วิธีแก้ที่ถูกต้อง: ต้องใช้ตัวหารที่เป็นเลขทศนิยม (Floating-point) เช่น 4095.0 หรือใช้ Type Casting (float)rawAnalog

5.3 Active-HIGH vs Active-LOW Relays & ฟังก์ชันควบคุมรีเลย์

โมดูลรีเลย์ทั่วไปในท้องตลาดส่วนใหญ่ทำงานแบบ Active-LOW (ใช้ LOW เพื่อสั่ง ON และ HIGH เพื่อสั่ง OFF) ในขณะที่รีเลย์บางรุ่นหรือการจำลองใน Wokwi เป็นแบบ Active-HIGH (ใช้ HIGH เพื่อ ON)

รูปแบบฟังก์ชันมาตรฐานระดับมืออาชีพ (Clean & Reusable): กำหนด Macro และสร้าง Helper Function setRelay() ช่วยให้สลับเปลี่ยนฮาร์ดแวร์ได้ง่ายโดยไม่ต้องแก้โค้ดหลัก:

// กำหนดระดับ Logic (true = Active-LOW, false = Active-HIGH)
#define RELAY_ACTIVE_LOW false

#if RELAY_ACTIVE_LOW
  #define RELAY_ON LOW
  #define RELAY_OFF HIGH
#else
  #define RELAY_ON HIGH
  #define RELAY_OFF LOW
#endif

// ฟังก์ชันควบคุมรีเลย์
void setRelay(uint8_t pin, bool state) {
  digitalWrite(pin, state ? RELAY_ON : RELAY_OFF);
}

// ใช้งานในโค้ดหลักได้อย่างชัดเจน ไม่สับสน:
setRelay(FAN_RELAY_PIN, true); // สั่งเปิดพัดลม
setRelay(FAN_RELAY_PIN, false); // สั่งปิดพัดลม

5.4 ตารางการสืบค้นและวินิจฉัยปัญหา (Troubleshooting Decision Matrix)

อาการผิดปกติ สาเหตุที่เป็นไปได้ วิธีแก้ไข
Serial Monitor แสดง nan หรือ Failed to read อ่านค่าเซ็นเซอร์เร็วกว่า 2 วินาที หรือสาย DATA หลวม เพิ่ม Delay/Non-blocking interval เป็น 2000ms+ และตรวจสายต่อขา 33
เปิดเครื่องแล้วพัดลมหมุนทันทีทั้งที่อุณหภูมิปกติ โมดูลรีเลย์เป็นแบบ Active-LOW ปรับค่าเริ่มต้นใน setup() เป็น HIGH และสลับคำสั่งเปิด/ปิด
ค่า ADC อ่านได้ 0 ถึง 4095 ไม่สม่ำเสมอ หรือเต็มเร็วมาก ไม่ได้ตั้งค่า ADC Attenuation 11dB บน ESP32 เพิ่มคำสั่ง analogSetPinAttenuation(36, ADC_11db); ใน setup()

6.1 รูปแบบคำสั่งใน Serial Monitor / Message Box

ไมโครคอนโทรลเลอร์สามารถรับข้อความคำสั่งจาก Serial Monitor Input Box (Baud rate 115200) หรือกล่องข้อความบนแดชบอร์ด เพื่อควบคุมและตั้งค่าพารามิเตอร์แบบทันที:

วัตถุประสงค์ คำสั่งที่ต้องพิมพ์ใน Message Box การตอบสนองของไมโครคอนโทรลเลอร์
สั่งเปิดพัดลม FAN_ON จ่ายไฟ GPIO 5 = HIGH, ตอบกลับ [OK] Fan turned ON
สั่งปิดพัดลม FAN_OFF จ่ายไฟ GPIO 5 = LOW, ตอบกลับ [OK] Fan turned OFF
ปรับเกณฑ์อุณหภูมิ SET_TEMP=32.0 อัปเดต tempThreshold = 32.0, ตอบกลับ [OK] Threshold updated
อ่านสถานะระบบ STATUS ส่งข้อความสรุปค่าเซ็นเซอร์และสถานะรีเลย์ปัจจุบัน
// ตัวอย่างโค้ดภาษา C++ ในการถอดรหัสข้อความจาก Serial Box:
if (Serial.available() > 0) {
  String cmd = Serial.readStringUntil('\n');
  cmd.trim(); // ตัดช่องว่างและตัวขึ้นบรรทัดใหม่ออก
  if (cmd == "FAN_ON") { digitalWrite(FAN_PIN, HIGH); }
  else if (cmd.startsWith("SET_TEMP=")) {
    tempThreshold = cmd.substring(9).toFloat();
  }
}

ข้อควรระวัง: ตรวจสอบการตั้งค่า Line Ending ใน Serial Monitor ให้ตรงกับ Newline (\n) และตั้ง Baud Rate เป็น 115200 เสมอ

คู่มือการเชื่อมต่อฮาร์ดแวร์จริงผ่าน Web Serial API

Web Serial API คืออะไร?

เป็นมาตรฐานเว็บสมัยใหม่ที่ช่วยให้หน้าเว็บสามารถสื่อสารกับไมโครคอนโทรลเลอร์ (ESP32 / ESP8266 / Arduino) ผ่านสาย USB ได้โดยตรง โดยไม่ต้องลงโปรแกรมเพิ่มเติม

ขั้นตอนการเชื่อมต่อบอร์ดจริง:

  1. เสียบสาย USB เชื่อมต่อบอร์ด ESP32 / ESP8266 เข้ากับคอมพิวเตอร์
  2. กดปุ่ม "เชื่อมต่อบอร์ดจริง (USB)" บนกล่อง Web Serial
  3. เบราว์เซอร์จะแสดงหน้าต่างเลือกพอร์ต ให้เลือก USB-Serial (เช่น CH340, CP2102 หรือ FTDI) แล้วกด Connect
  4. หน้าต่าง Terminal จะเริ่มสตรีมข้อมูลค่าเซ็นเซอร์และสถานะรีเลย์แบบ Real-Time อัตโนมัติ (115200 baud)

เบราว์เซอร์ที่รองรับ:

✓ รองรับ: Google Chrome, Microsoft Edge, Opera, Brave (บน Windows, macOS, Linux, ChromeOS)

Safari และ Firefox ยังไม่รองรับ Web Serial API
