ใบงานที่ 3.1: การพัฒนา IoT Web Dashboard ด้วย LittleFS และ REST API

การพัฒนา Embedded HTTP Web Server ด้วย Flash Memory (LittleFS) และ REST API เชื่อมต่อเซ็นเซอร์/รีเลย์

ระบบส่งใบงานปฏิบัติการออนไลน์
ออนไลน์ (พร้อมส่งรายงาน)
คู่มือใบงานการทดลอง

วัตถุประสงค์การเรียนรู้

  • เข้าใจสถาปัตยกรรม Client-Server และหลักการทำงานของ HTTP Web Server บนพอร์ต 80
  • สามารถจัดเก็บไฟล์ HTML, CSS, JavaScript ลงใน Flash Memory ด้วยระบบไฟล์ LittleFS
  • สามารถพัฒนา REST API Endpoint (`/api/data` และ `/api/control`) เพื่อส่งค่าอุณหภูมิ/ความชื้นจาก DHT11 และสัญญาณอนาล็อก (GPIO 36)
  • เข้าใจการสตรีมไฟล์ด้วย server.streamFile() เพื่อป้องกันปัญหา RAM หมด (Heap Overflow)
  • สามารถสร้างตรรกะควบคุมอัตโนมัติ 2 เงื่อนไข (อุณหภูมิควบคุม Relay 1 และสัญญาณอนาล็อกควบคุม Relay 2) ควบคู่กับ Web Interface

การเชื่อมต่อวงจรฮาร์ดแวร์ (ใช้วงจรเดิมจาก Lab 1)

อุปกรณ์ / โมดูล ขา ESP32 / ESP8266 โหมดการทำงาน หน้าที่การทำงาน
DHT11 Sensor (Temp & Humidity) GPIO 33 (ESP8266: D3/0) DIGITAL INPUT วัดค่าอุณหภูมิ (°C) และความชื้นสัมพัทธ์ (%RH)
Analog Sensor (Potentiometer / VR) GPIO 36 (VP / A0) ANALOG INPUT อ่านสัญญาณแรงดันอนาล็อก (0–4095)
Relay 1 Module (พัดลมระบายความร้อน) GPIO 5 (ESP8266: D7/13) OUTPUT (Active-LOW) ควบคุมพัดลมตามเงื่อนไขอุณหภูมิ DHT11 / สั่งผ่าน Web
Relay 2 Module (ปั๊มพ่นหมอก / รดน้ำ) GPIO 23 (ESP8266: D0/16) OUTPUT (Active-LOW) ควบคุมปั๊มหมอกตามเงื่อนไขสัญญาณอนาล็อก / สั่งผ่าน Web
Status LED (Onboard LED) GPIO 2 OUTPUT แสดงสถานะการเชื่อมต่อ Wi-Fi และการเรียก API

สถาปัตยกรรม LittleFS และ REST API Endpoints

URI Endpoint / พาธ ชนิด / MIME Type คำอธิบายการทำงาน
/ หรือ /index.html text/html (LittleFS) หน้าหลัก IoT Web Dashboard ควบคุมและมอนิเตอร์บอร์ด
/styles.css & /app.js text/css, application/javascript ไฟล์ตกแต่ง Glassmorphism Theme และสคริปต์ Polling API
/api/data application/json (Dynamic API) อ่านค่าเซ็นเซอร์ (temp, hum, analog) และสถานะรีเลย์ (relay1, relay2)
/api/control application/json (Control API) รับคำสั่งเปิด-ปิด Relay 1 (พัดลม) และ Relay 2 (ปั๊มหมอก) ผ่าน Web Client
อื่นๆ (URI ที่ไม่มีไฟล์) text/plain ดักจับด้วย server.onNotFound() ส่งรหัส 404

#include <Arduino.h>
#if defined(ESP8266)
  #include <ESP8266WiFi.h>
  #include <ESP8266WebServer.h>
  ESP8266WebServer server(80);
#elif defined(ESP32)
  #include <WiFi.h>
  #include <WebServer.h>
  WebServer server(80);
#endif
#include <LittleFS.h>
#include <DHT.h>

const char* ssid = "iot_512";
const char* password = "iot123456";

#define DHTPIN        33   // ขาข้อมูล DHT11 จากวงจร Lab 1
#define DHTTYPE       DHT11
#define ANALOG_PIN    36   // สัญญาณอนาล็อก (Potentiometer / LDR)
#define RELAY1_PIN     5   // Relay 1 พัดลมระบายความร้อน (Active-LOW)
#define RELAY2_PIN    23   // Relay 2 ปั๊มพ่นหมอก (Active-LOW)

DHT dht(DHTPIN, DHTTYPE);

bool relay1State = false;
bool relay2State = false;

// ฟังก์ชัน REST API ส่งค่าเซ็นเซอร์ DHT11, อนาล็อก และสถานะรีเลย์แบบ JSON
void handleApiData() {
  float tempC = dht.readTemperature();
  float hum = dht.readHumidity();
  int rawAnalog = analogRead(ANALOG_PIN);
  
  if (isnan(tempC)) tempC = 0.0;
  if (isnan(hum)) hum = 0.0;
  
  String json = "{";
  json += "\"mac\":\"" + WiFi.macAddress() + "\",";
  json += "\"temp\":" + String(tempC, 1) + ",";
  json += "\"hum\":" + String(hum, 1) + ",";
  json += "\"analog\":" + String(rawAnalog) + ",";
  json += "\"relay1\":" + String(relay1State ? "true" : "false") + ",";
  json += "\"relay2\":" + String(relay2State ? "true" : "false");
  json += "}";
  server.send(200, "application/json", json);
}

// ฟังก์ชันสตรีมไฟล์หน้าเว็บจาก LittleFS
void handleFileRequest() {
  String path = server.uri();
  if (path.endsWith("/")) path += "index.html";
  String dataType = "text/plain";
  if (path.endsWith(".html")) dataType = "text/html";
  else if (path.endsWith(".css")) dataType = "text/css";
  else if (path.endsWith(".js")) dataType = "application/javascript";

  // ตรวจสอบว่ามีไฟล์ใน LittleFS Flash Memory หรือไม่
  if () {
    File file = LittleFS.open(path, "r");
    // สตรีมไฟล์ไปยัง Client โดยไม่กิน RAM Heap
    ;
    file.close();
  } else {
    server.send(404, "text/plain", "404: File Not Found");
  }
}

void setup() {
  Serial.begin(115200);
  dht.begin();
  pinMode(RELAY1_PIN, OUTPUT);
  pinMode(RELAY2_PIN, OUTPUT);
  digitalWrite(RELAY1_PIN, HIGH); // ปิดรีเลย์เริ่มต้น (Active-LOW)
  digitalWrite(RELAY2_PIN, HIGH);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) delay(500);
  Serial.print("Web Server IP: "); Serial.println(WiFi.localIP());
  Serial.print("MAC Address: "); Serial.println(WiFi.macAddress());

  // 1. เมานต์ระบบไฟล์ LittleFS บน Flash Memory
  if (!) {
    Serial.println("Mount Failed");
    return;
  }

  // 2. กำหนด REST API Route สำหรับอ่านค่าเซ็นเซอร์
  ;

  // 3. กำหนด Catch-all Handler สำหรับไฟล์หน้าเว็บ
  server.onNotFound(handleFileRequest);
  server.begin();
}

void loop() {
  // 4. ประมวลผลคำขอ HTTP Request จาก Web Browser
  ;
  delay(2);
}

โจทย์ท้าทาย (Challenge Mission)

ให้นักศึกษาต่อยอดจากโครงร่างโค้ดตั้งต้นด้านบน ให้กลายเป็น Interactive Smart Environment Control Dashboard แบบ 2 ทิศทาง:

  • 1. เงื่อนไขอุณหภูมิควบคุม Relay 1 (พัดลม): เขียนตรรกะตรวจสอบค่าอุณหภูมิจาก DHT11 (`tempC`) หากอุณหภูมิ ≥ 35.0 °C ให้สั่งเปิด Relay 1 (GPIO 5, Active-LOW) อัตโนมัติเพื่อระบายความร้อน และเมื่ออุณหภูมิลดลงต่ำกว่า 32.0 °C ให้สั่งปิด
  • 2. เงื่อนไขสัญญาณอนาล็อกควบคุม Relay 2 (ปั๊มหมอก/รดน้ำ): ตรวจสอบค่าสัญญาณอนาล็อกจาก GPIO 36 (`rawAnalog`) หากค่า > 2500 ให้สั่งเปิด Relay 2 (GPIO 23, Active-LOW) อัตโนมัติ และเมื่อค่า ≤ 2000 ให้สั่งปิด
  • 3. REST API ควบคุม Manual Override (`/api/control`): พัฒนา Route รองรับคำสั่งเปิด/ปิด Relay 1 และ Relay 2 ผ่านเว็บ (เช่น /api/control?relay=1&state=1 หรือ relay=2)
  • 4. Web Polling ใน LittleFS (`data/app.js`): ดึง fetch('/api/data') ทุก 1 วินาที เพื่ออัปเดตค่า Temperature (°C), Humidity (%), Analog Value, และสถานะ Relay 1 & Relay 2 สดแบบเรียลไทม์

เนื้อหาทฤษฎี: Embedded Web Server & LittleFS

1.1 วงจรคำขอ-ตอบกลับ (HTTP Request-Response Cycle)

HTTP (HyperText Transfer Protocol) ทำงานบนพอร์ตมาตรฐาน Port 80 ด้วยสถาปัตยกรรมแบบ Stateless Client-Server เมื่อผู้ใช้พิมพ์ IP Address ของบอร์ด ESP32 ลงในเบราว์เซอร์ เบราว์เซอร์จะส่งคำขอ GET / HTTP/1.1 เข้ามายังไมโครคอนโทรลเลอร์ เซิร์ฟเวอร์จะประมวลผลและส่ง HTTP Status Code (เช่น 200 OK หรือ 404 Not Found) พร้อมทั้งข้อมูลไฟล์กลับไป

Client Browser --- [ GET /index.html HTTP/1.1 ] ---> ESP32 Web Server
Client Browser <-- [ HTTP/1.1 200 OK + HTML ] --- ESP32 Web Server

2.1 ระบบไฟล์ LittleFS สำหรับ Flash Memory

LittleFS คือระบบแฟ้มข้อมูลขนาดกะทัดรัดที่ถูกออกแบบมาเพื่อชิป Flash Memory ที่มีหน่วยความจำจำกัดโดยเฉพาะ มีข้อดีที่เหนือกว่า SPIFFS เดิมอย่างมาก ได้แก่:

  • Power-loss Resilience: ป้องกันข้อมูลเสียหายเมื่อเกิดไฟตกหรือถอดปลั๊กกะทันหัน
  • Dynamic Wear Leveling: กระจายการเขียนข้อมูลไปยังบล็อกต่างๆ ของ Flash อย่างทั่วถึง ยืดอายุการใช้งานฮาร์ดแวร์
  • Directory Support: รองรับโครงสร้างโฟลเดอร์ย่อยได้อย่างสมบูรณ์
  • ประหยัด RAM: ใช้ขนาด Buffer ในการทำงานน้อยมาก

3.1 การสตรีมไฟล์ขนาดใหญ่ด้วย server.streamFile()

เนื่องจากไมโครคอนโทรลเลอร์มีหน่วยความจำ RAM (Heap) จำกัด หากเราอ่านไฟล์ทั้งไฟล์เข้ามาเก็บในตัวแปร String แล้วสั่ง server.send() ไฟล์ขนาดใหญ่ เช่น รูปภาพหรือไฟล์สไตล์ CSS จะทำให้ RAM หมดและบอร์ดรีสตาร์ตทันที การใช้ server.streamFile(file, contentType) จะทำการอ่านข้อมูลทีละบล็อก (Chunk) แล้วส่งตรงไปยัง Client ทันที ทำให้ประหยัด RAM อย่างมหาศาล

// ตัวอย่างการใช้งาน streamFile
File file = LittleFS.open("/index.html", "r");
server.streamFile(file, "text/html");
file.close();

4.1 รูปแบบข้อความ JSON ใน REST API & Request Body

เมื่อผู้ใช้หรือเว็บแอปส่งคำสั่งผ่าน HTTP Request Body / Test Console ข้อมูลจะต้องอยู่ในรูปแบบ JSON ที่ถูกต้อง:

Endpoint & Method สิ่งที่ต้องพิมพ์ใน Request Body ผลตอบกลับจาก ESP32 / ESP8266
GET /api/status (ไม่มี Body) {"temp": 28.5, "led": false, "heap": 215400}
POST /api/led {"state": true} เปิด Onboard LED, ตอบกลับ {"success": true, "led": true}
POST /api/relay {"relay": true, "timer": 10} สั่งเปิดรีเลย์ 10 วินาที, ตอบกลับ {"success": true}

⚠️ ข้อควรระวังเรื่องไวยากรณ์ JSON (JSON Syntax Rules):

  • Key ต้องอยู่ในเครื่องหมาย Double Quotes "..." เท่านั้น (ห้ามใช้ Single Quote '...')
  • ค่า Boolean ต้องพิมพ์ true หรือ false (ตัวพิมพ์เล็ก ไม่มีเครื่องหมายคำพูด)
  • ค่าตัวเลข เช่น 28.5 ไม่ต้องใส่เครื่องหมายคำพูด
