วัตถุประสงค์

  1. เพื่อเรียนรู้และบูรณาการสถาปัตยกรรม Dual-Mode IoT System โดยรันทั้ง Local WebSocket Server (Port 81) และ Cloud MQTT Client (Port 1883) บนบอร์ด ESP32/ESP8266 พร้อมกันในโปรแกรมเดียว
  2. เพื่อแก้ปัญหาความหน่วงและอินเทอร์เน็ตล่ม โดยให้ระบบสามารถสั่งงานได้แบบ 0-Latency บนวง LAN และสั่งงานข้ามทวีปได้เมื่อมีอินเทอร์เน็ต

#include <WiFi.h>
#include <WebServer.h>
#include <WebSocketsServer.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <LittleFS.h>
#include <DHT.h>

#define DHTPIN 33
#define DHTTYPE DHT11
#define FAN_PIN 19   // Relay 1
#define MIST_PIN 23  // Relay 2
#define SW1_PIN 0    // Switch 1 (PULLUP)
#define SW2_PIN 5    // Switch 2 (PULLUP)

DHT dht(DHTPIN, DHTTYPE);
WebServer server(80);
WebSocketsServer webSocket = WebSocketsServer(  );

WiFiClient espClient;
PubSubClient mqttClient(  );

bool fanState = false;
bool mistState = false;

// ⚡ ฟังก์ชัน Broadcast อัปเดตสถานะแบบ Real-Time ทั้ง Local WebSocket และ Cloud MQTT
void broadcastHybridState() {
  float t = dht.readTemperature();
  float h = dht.readHumidity();
  
  JsonDocument doc;
  doc["temp"] = isnan(t) ? 0 : t;
  doc["humidity"] = isnan(h) ? 0 : h;
  doc["fan"] = fanState ? 1 : 0;
  doc["mist"] = mistState ? 1 : 0;
  doc["mqtt"] = mqttClient.connected();
  
  String payload;
  serializeJson(doc, payload);
  
  // 1. ส่งข้อมูลไปยัง Local WebSocket Clients ทุกเครื่องบน LAN พร้อมกัน
  webSocket.broadcastTXT(payload);
  
  // 2. ส่งข้อมูลขึ้น Cloud MQTT Broker
  if (mqttClient.connected()) {
    mqttClient.publish(  , payload.c_str());
  }
}

// 🌐 ฟังก์ชันจัดการคำสั่งจาก Cloud MQTT
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  JsonDocument doc;
  deserializeJson(doc, payload, length);
  if (doc["action"] == "toggle_fan") {
    fanState = doc["value"];
    digitalWrite(FAN_PIN, fanState ? HIGH : LOW);
    broadcastHybridState(); // ⚡ ซิงค์สถานะใหม่กลับทั้ง WebSocket และ MQTT
  }
}

// ⚡ ฟังก์ชันจัดการคำสั่งจาก Local WebSocket
void webSocketEvent(uint8_t num, WStype_t type, uint8_t * payload, size_t length) {
  if (type == WStype_TEXT) {
    JsonDocument doc;
    deserializeJson(doc, payload, length);
    if (doc["action"] == "toggle_fan") {
      fanState = doc["value"];
      digitalWrite(FAN_PIN, fanState ? HIGH : LOW);
      broadcastHybridState(); // ⚡ ซิงค์สถานะใหม่กลับทั้ง WebSocket และ MQTT
    }
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(FAN_PIN, OUTPUT);
  pinMode(MIST_PIN, OUTPUT);
  pinMode(SW1_PIN, INPUT_PULLUP);
  pinMode(SW2_PIN, INPUT_PULLUP);
  dht.begin();
  LittleFS.begin(true);
  
  WiFi.begin("iot_512", "iot123456");
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  
  server.begin();
  webSocket.begin();
  webSocket.onEvent(  );
  
  mqttClient.setServer("broker.emqx.io", 1883);
  mqttClient.setCallback(mqttCallback);
}

void loop() {
  server.handleClient();
  webSocket.loop();
  
  // Non-blocking MQTT Reconnection (ไม่ใช้ delay เพื่อไม่ให้บล็อก WebSocket)
  static unsigned long lastMqttCheck = 0;
  if (!mqttClient.connected() && (millis() - lastMqttCheck > 5000)) {
    lastMqttCheck = millis();
    String clientId = "ESP32-Hybrid-" + String((uint32_t)ESP.getEfuseMac(), HEX);
    if (mqttClient.connect(clientId.c_str())) {
      mqttClient.subscribe(  );
    }
  }
  if (mqttClient.connected()) { mqttClient.loop(); }

  // Periodic Telemetry: ส่งค่าเซ็นเซอร์ตามรอบเวลาปกติทุก 5 วินาที
  static unsigned long lastMsg = 0;
  if (millis() - lastMsg > 5000) {
    lastMsg = millis();
    broadcastHybridState();
  }
}

ทฤษฎีที่เกี่ยวข้อง: การเชื่อมต่อและควบคุมผ่าน Cloud MQTT

1.1 หลักการทำงานของ Publish / Subscribe

MQTT (Message Queuing Telemetry Transport) เป็นโปรโตคอลมาตรฐานสำหรับ IoT แบบ Lightweight ที่ทำงานผ่านตัวกลางเรียกว่า Broker:

  • Publisher: อุปกรณ์ที่ส่งข้อมูลขึ้นหัวข้อ (Topic) เช่น บอร์ด ESP32 ส่งค่าอุณหภูมิ
  • Subscriber: อุปกรณ์ที่รอรับข้อมูลจากหัวข้อที่สนใจ เช่น Dashboard หรือแอปพลิเคชันมือถือ
  • Broker: เซิร์ฟเวอร์ทำหน้าที่คัดแยกและกระจายข้อความ (เช่น broker.emqx.io)

2.1 โครงสร้างหัวข้อ (Topic) และระดับคุณภาพ (QoS)

การตั้งชื่อ Topic นิยมแบ่งระดับชั้นด้วยเครื่องหมาย Slash / เช่น:

esp32-climate-node/sensor/temperature
esp32-climate-node/control/fan/cmd

สัญลักษณ์ Wildcard ในการ Subscribe:

  • + (Single-level Wildcard): แทนที่ 1 ระดับชั้น เช่น sensor/+/temp
  • # (Multi-level Wildcard): แทนที่ทุกระดับชั้นที่เหลือ เช่น sensor/#

ระดับ QoS (Quality of Service):

  • QoS 0 : At most once (ส่งครั้งเดียว ไม่มีการยืนยัน เหมาะสำหรับ Sensor Telemetry)
  • QoS 1 : At least once (ส่งซ้ำจนกว่าจะได้รับยืนยัน มั่นใจว่าถึงแน่นอน)
  • QoS 2 : Exactly once (การันตีถึงแน่นอนและได้รับเพียงครั้งเดียว เหมาะกับคำสั่งฉุกเฉินหรือธุรกรรม)

3.1 MQTT Keep-Alive, Ping และ Last Will & Testament (LWT)

Keep-Alive Interval: เป็นช่วงเวลาที่ Client และ Broker จะส่งข้อความ PINGREQ / PINGRESP เพื่อยืนยันว่าการเชื่อมต่อยังคงทำงานอยู่ หากขาดการส่งภายในเวลาที่กำหนด Broker จะถือว่าไคลเอนต์หลุดการเชื่อมต่อ

Last Will & Testament (LWT): ข้อความพินัยกรรมที่กำหนดไว้ล่วงหน้า เมื่อไคลเอนต์หลุดกะทันหัน Broker จะ Publish ข้อความนี้อัตโนมัติ เช่น แจ้งสถานะ {"status":"offline"}

4.1 รูปแบบข้อความ JSON Payload & ArduinoJson

การส่งข้อมูลแบบ JSON ช่วยให้รวมหลายค่าเซ็นเซอร์และสถานะควบคุมไว้ใน Payload เดียวอย่างมีประสิทธิภาพ:

Topic (หัวข้อ) ข้อความ JSON Payload ผลลัพธ์บนอุปกรณ์ IoT
esp-node/control/cmd {"action":"toggle_fan","value":true} สั่งเปิดรีเลย์พัดลม GPIO 5
esp-node/state {"temp":28.5,"hum":65.0,"soil":45.2,"fan":1,"press":3} รายงานค่า Telemetry ทุก 5 วินาที
// ตัวอย่างการ Subscribe และถอดรหัสใน C++ (PubSubClient + ArduinoJson):
void callback(char* topic, byte* payload, unsigned int length) {
  JsonDocument doc;

  if (doc["action"] == "toggle_fan") {

5.1 การตรวจวินิจฉัยและแก้ปัญหา Client ID Collision

บน Public Broker ทุกๆ การเชื่อมต่อจะต้องมี Client ID ที่ไม่ซ้ำกับใครในโลก:

  • อาการ Client Collision: เมื่อใช้ Client ID ซ้ำกัน Broker จะ Disconnect ไคลเอนต์ตัวแรกออก แล้วสลับต่อตัวใหม่ เกิดการวนลูป Reconnect ต่อเนื่อง
  • แนวทางแก้ไขที่ดี: ดึง Hardware MAC Address ของชิปมาต่อท้าย เช่น "ESP32-" + String((uint32_t)ESP.getEfuseMac(), HEX) หรือใส่เลขรหัสนักศึกษา
  • หลีกเลี่ยงการใช้ delay() บล็อกฟังก์ชัน loop(): เพื่อให้ mqttClient.loop() ประมวลผล Network Packet และ Keep-Alive อย่างต่อเนื่อง

Web Serial API คืออะไร?

เป็นมาตรฐานเว็บสมัยใหม่ที่ช่วยให้หน้าเว็บสามารถสื่อสารกับไมโครคอนโทรลเลอร์ (ESP32 / ESP8266 / Arduino) ผ่านสาย USB ได้โดยตรง โดยไม่ต้องลงโปรแกรมเพิ่มเติม

ขั้นตอนการเชื่อมต่อบอร์ดจริง:

  1. เสียบสาย USB เชื่อมต่อบอร์ด ESP32 / ESP8266 เข้ากับคอมพิวเตอร์
  2. กดปุ่ม "เชื่อมต่อบอร์ดจริง (USB)" บนกล่อง Web Serial
  3. เบราว์เซอร์จะแสดงหน้าต่างเลือกพอร์ต ให้เลือก USB-Serial (เช่น CH340, CP2102 หรือ FTDI) แล้วกด Connect
  4. หน้าต่าง Terminal จะเริ่มสตรีมข้อความ Serial Log และ JSON Payload จากบอร์ดจริงทันที

ข้อควรระวังและวิธีแก้ปัญหา:

  • เบราว์เซอร์ที่รองรับ: Google Chrome, Microsoft Edge, Opera, Brave (ไม่รองรับ Safari และ Firefox)
  • ปัญหาหาพอร์ตไม่เจอ: ตรวจสอบว่าได้ติดตั้ง Driver ชิป USB-to-UART ของบอร์ดแล้วหรือไม่ (เช่น CH340 Driver หรือ CP210x Driver)
  • ปัญหา Port Busy (เปิดไม่ขึ้น): ตรวจสอบว่าไม่ได้เปิดโปรแกรมอื่นที่ใช้พอร์ต COM ซ้ำกันอยู่ เช่น Arduino IDE Serial Monitor, PlatformIO Serial Monitor หรือ Putty
  • ตัวอักษรภาษาต่างดาว: เลือก Baud Rate ให้ตรงกับ Serial.begin(...) ในโค้ดของบอร์ด (เช่น 115200)
