ใบงานที่ 4.1: การควบคุมอุปกรณ์และการอ่านค่าเซ็นเซอร์ผ่าน HTTP GET / POST

การพัฒนา REST API Web Server บน ESP32 / ESP8266 เชื่อมโยงอุปกรณ์จากใบงานที่ 1 และ 2

วัตถุประสงค์การเรียนรู้

  • เข้าใจความแตกต่างและการทำงานของเมธอด HTTP GET (อ่านค่า) และ HTTP POST (ส่งคำสั่งควบคุม)
  • สร้าง Web Server บน ESP32/ESP8266 ให้บริการ REST API Endpoints: GET /api/data และ POST /api/relay
  • เชื่อมโยงอุปกรณ์ DHT11, รีเลย์พัดลม, ปุ่มกดสวิตช์ GPIO 0, และจอแสดงผล OLED SSD1306 (I2C)
  • สามารถพัฒนา Client ส่ง HTTP Request เพื่ออ่านค่าและควบคุมอุปกรณ์ พร้อมจัดการ JSON Payload ด้วย ArduinoJson v7

โครงร่างโค้ดโปรแกรม (REST API Server Skeleton)

lab4_1_rest_server.ino 
#if defined(ESP8266)
#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
ESP8266WebServer server(80);
#define DHTPIN 2
#define FAN_RELAY_PIN 13
#elif defined(ESP32)
#include <WiFi.h>
#include <WebServer.h>
WebServer server(80);
#define DHTPIN 33
#define FAN_RELAY_PIN 5
#endif
#include <ArduinoJson.h>
#include <DHT.h>

DHT dht(DHTPIN, DHT11);
bool relayState = false;

// 1. ส่งค่าเซ็นเซอร์กลับเป็น JSON เมื่อได้รับ HTTP GET /api/data
void handleGetData() {
  JsonDocument doc;
  doc["temp"] = dht.readTemperature();
  doc["humidity"] = dht.readHumidity();
  doc["relay"] = relayState;
  
  String jsonStr;
  ;
  ;
}

// 2. รับคำสั่งควบคุมรีเลย์เมื่อได้รับ HTTP POST /api/relay
void handleSetRelay() {
  if (server.hasArg("plain")) {
    JsonDocument doc;
    ;
    if (doc["relay"].is<JsonVariant>()) {
      relayState = doc["relay"];
      digitalWrite(FAN_RELAY_PIN, relayState ? HIGH : LOW);
    }
    server.send(200, "application/json", "{\"success\":true}");
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(FAN_RELAY_PIN, OUTPUT);
  dht.begin();
  WiFi.begin("iot_512", "iot123456");
  while (WiFi.status() != WL_CONNECTED) delay(500);

  // ลงทะเบียน REST API Endpoints
  ;
  ;

  server.begin();
}

void loop() {
  server.handleClient();
}

เนื้อหาทฤษฎี: REST API & HTTP GET / POST บน ESP32 / ESP8266

1.1 หลักการของ HTTP GET vs HTTP POST

HTTP GET: ใช้สำหรับการร้องขออ่านข้อมูลจากเซิร์ฟเวอร์ โดยไม่มีผลข้างเคียง (Side Effects / Idempotent) ต่อสถานะของฮาร์ดแวร์ เช่น การเรียก GET /api/data เพื่ออ่านอุณหภูมิและความชื้น

HTTP POST: ใช้สำหรับการส่งชุดข้อมูล (Payload) ไปยังเซิร์ฟเวอร์เพื่อเปลี่ยนแปลงสถานะของฮาร์ดแวร์ (State Mutation) เช่น การส่ง POST /api/relay พร้อม Body {"relay": true} เพื่อสั่งเปิดพัดลม

// การลงทะเบียน Endpoint ใน C++:
server.on("/api/data", HTTP_GET, handleGetData);
server.on("/api/relay", HTTP_POST, handleSetRelay);

2.1 การจัดการ JSON ด้วย ArduinoJson v7

ใน ArduinoJson v7 เราใช้คลาส JsonDocument เพียงตัวเดียวในการจัดการทั้งการอ่านและการเขียน โดยระบบจะจัดสรรหน่วยความจำแบบไดนามิก:

// 1. Serialization (แปลงข้อมูลเป็น JSON String ส่งกลับ Client):
JsonDocument doc;
doc["temp"] = temperature;
doc["relay"] = relayState;
String jsonStr;
serializeJson(doc, jsonStr);
server.send(200, "application/json", jsonStr);

// 2. Deserialization (ถอดรหัส JSON จาก Client Payload):
deserializeJson(doc, server.arg("plain"));
bool newState = doc["relay"];

3.1 HTTP Polling และปัญหา Cross-Origin (CORS)

HTTP Polling: หน้าเว็บเรียกใช้ fetch('/api/data') ซ้ำๆ ทุก 1-2 วินาที แม้จะเขียนง่าย แต่สร้าง Overhead จาก Header ทุกครั้ง

CORS Header: หากหน้าเว็บ Dashboard รันอยู่คนละ Host/Port กับ ESP32 จะต้องเพิ่ม Header ตอบกลับเพื่อไม่ให้เบราว์เซอร์บล็อกคำขอ:

server.sendHeader("Access-Control-Allow-Origin", "*");
server.sendHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");

4.1 รูปแบบ Payload ในการเรียก HTTP POST & Postman / Fetch Console

เมื่อส่งคำสั่งควบคุมผ่าน HTTP POST /api/relay ทั้งผ่าน JavaScript fetch(), Postman, หรือ Serial Test Console จะต้องส่ง JSON Body ดังนี้:

คำสั่งที่ต้องการ สิ่งที่ต้องพิมพ์ใน Request Body (JSON) ผลตอบกลับจากเซิร์ฟเวอร์
สั่งเปิดพัดลม {"relay": true} {"success": true, "relay": true, "message": "Fan ON"}
สั่งปิดพัดลม {"relay": false} {"success": true, "relay": false, "message": "Fan OFF"}
สลับสถานะ (Toggle) {"action": "toggle"} {"success": true, "relay": [สถานะใหม่]}
// ตัวอย่างการส่งคำสั่งด้วย JavaScript fetch():
fetch('/api/relay', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ relay: true })
}).then(res => res.json()).then(data => console.log(data));
