ใบงานเสริม: การตั้งค่า Wi-Fi ผ่านหน้าเว็บ Web Configuration

เรียนรู้วิธีการทำงานของ Captive Portal การเขียนโปรแกรมทำหน้าเว็บต้อนรับของอุปกรณ์ IoT และการบันทึกข้อมูลแบบถาวรในระบบไฟล์ LittleFS

วัตถุประสงค์

  1. เพื่อเข้าใจหลักการทำงานของ Captive Portal และ DNS Server ในการเปลี่ยนเส้นทางผู้ใช้
  2. เพื่อสร้าง Web Server บน ESP32 / ESP8266 สำหรับรับค่า SSID และ Password ผ่านหน้าเว็บ
  3. เพื่อจัดเก็บบันทึกข้อมูลเครือข่ายลงใน Flash Memory ผ่าน LittleFS JSON อย่างถาวร

อุปกรณ์ที่ต้องใช้

รายการอุปกรณ์ รายละเอียด จำนวน
บอร์ดไมโครคอนโทรลเลอร์ ESP32 (IPST-WiFi) หรือ ESP8266 (AX-WiFi / NodeMCU) 1 บอร์ด
สาย USB USB-to-Serial Data Cable สำหรับอัพโหลดและดู Serial Output 1 เส้น
สมาร์ทโฟน หรือ คอมพิวเตอร์ สำหรับเชื่อมต่อ Wi-Fi (ESP-Config-AP) เพื่อทดสอบ Captive Portal 1 เครื่อง
เครือข่าย Wi-Fi เป้าหมาย Access Point 2.4 GHz ที่ต้องการให้บอร์ดเชื่อมต่อ 1 เครือข่าย
โครงร่างโค้ดโปรแกรมหลัก (Fill in the Blanks)

จงเติมคำสั่งในช่องว่างให้สมบูรณ์ เพื่อให้ DNS Server ชี้ปลายทางมายังอุปกรณ์ได้ถูกต้อง และเปิดเขียนไฟล์ตั้งค่า Wi-Fi ลงใน LittleFS ได้อย่างถูกต้อง 

#include <Arduino.h>
#include <Wire.h>
#include <LittleFS.h>
#include <ArduinoJson.h>
#include <DNSServer.h>

#if defined(ESP8266)
#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#else
#include <WiFi.h>
#include <WebServer.h>
#endif

#define BUTTON_PIN 0

const char* apSSID = "ESP-Config-AP";
const byte DNS_PORT = 53;
IPAddress apIP(192, 168, 4, 1);

DNSServer dnsServer;
#if defined(ESP8266)
ESP8266WebServer server(80);
#else
WebServer server(80);
#endif

bool apMode = false;
String wifiSSID = "";
String wifiPassword = "";

bool loadWifiConfig() {
  if (!LittleFS.exists("/config.json")) return false;
  File file = LittleFS.open("/config.json", "r");
  if (!file) return false;
  JsonDocument doc;
  deserializeJson(doc, file);
  file.close();
  wifiSSID = doc["ssid"].as<String>();
  wifiPassword = doc["pass"].as<String>();
  return true;
}

void saveWifiConfig(String ssid, String pass) {
  // เติมคำสั่งเปิดไฟล์เพื่อเขียนข้อมูลตั้งค่าเครือข่าย Wi-Fi
  File file = ;
  
  if (file) {
    JsonDocument doc;
    doc["ssid"] = ssid;
    doc["pass"] = pass;
    serializeJson(doc, file);
    file.close();
    Serial.println("Config saved successfully.");
  }
}

void handlePortal() {
  if (LittleFS.exists("/index.html")) {
    File file = LittleFS.open("/index.html", "r");
    server.streamFile(file, "text/html");
    file.close();
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  LittleFS.begin();
  
  if (loadWifiConfig()) {
    WiFi.mode(WIFI_STA);
    WiFi.begin(wifiSSID.c_str(), wifiPassword.c_str());
    int retries = 0;
    while (WiFi.status() != WL_CONNECTED && retries < 30) {
      delay(500);
      retries++;
    }
    if (WiFi.status() == WL_CONNECTED) return;
  }
  
  // Launch Captive Portal if connection failed
  apMode = true;
  WiFi.mode(WIFI_AP);
  WiFi.softAP(apSSID);
  dnsServer.start(DNS_PORT, "*", apIP);
  server.on("/", HTTP_GET, handlePortal);
  server.begin();
}

void loop() {
  if (apMode) {
    // เติมคำสั่งประมวลผล DNS เพื่อดักจับทุกคำขอชี้ทางมายังอุปกรณ์
    ;
    
    server.handleClient();
  }
  delay(1);
}

ทฤษฎีที่เกี่ยวข้อง: การตั้งค่า Wi-Fi ผ่านหน้าเว็บ Web Configuration (Captive Portal)

1.1 กลไกการทำงานของ Captive Portal

เมื่อบอร์ด ESP ไม่สามารถเชื่อมต่อกับ Router หลักได้ บอร์ดจะสลับเข้าสู่ Fallback Access Point Mode และเปิดบริการ DNS Server บนพอร์ต 53:

  • เมื่อสมาร์ทโฟนหรือคอมพิวเตอร์เชื่อมต่อ Wi-Fi และพยายามเปิดเว็บใดๆ DNS Server จะดักจับคำขอ (Wildcard *) และตอบกลับด้วยไอพีของบอร์ดเอง (192.168.4.1) เสมอ
  • ระบบปฏิบัติการของมือถือจะตรวจพบและเปิดหน้าต่างล็อกอิน / หน้าเว็บตั้งค่าให้ผู้ใช้กรอก SSID และ Password ทันทีโดยไม่ต้องพิมพ์ URL

2.1 การบันทึกค่าลง Flash Memory อย่างถาวร

เมื่อผู้ใช้กดบันทึกผ่านหน้าเว็บ Web Server (Port 80) จะรับค่าผ่านคำขอ HTTP POST แล้วบันทึกลงไฟล์ /config.json ในระบบไฟล์ LittleFS จากนั้นสั่ง ESP.restart() เพื่อรีบูตบอร์ดเข้าสู่ Station Mode พร้อมข้อมูลเครือข่ายใหม่ทันที
