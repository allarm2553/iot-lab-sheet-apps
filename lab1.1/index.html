วัตถุประสงค์

  1. เพื่อเรียนรู้การเชื่อมต่อและอ่านค่าสวิตช์ปุ่มกด (Push Button) ร่วมกับบอร์ดไมโครคอนโทรลเลอร์
  2. เพื่อเข้าใจปัญหาสัญญาณรบกวนหน้าสัมผัสของสวิตช์ (Switch Bouncing) และวิธีการแก้ไขด้วยซอฟต์แวร์ (Debounce)
  3. เพื่อเรียนรู้การประมวลผลอินพุตแบบตรวจสอบการกดสั้นเพื่อสลับสถานะ (Toggle) และการกดค้าง (Long Press) เพื่อสั่งงานรีเลย์

โค้ดโครงร่าง (Skeleton Code)

ศึกษาโค้ดต่อไปนี้และค้นหาคำตอบเพื่อนำมาเติมในช่องว่างทางขวามือให้สมบูรณ์:

lab1_1_skeleton.ino 
#include <Arduino.h>

#define BUTTON_PIN 0  // (GPIO 0: SW1 บน IPST-WiFi หรือ D3/FLASH บน AX-WiFi)
#define RELAY_PIN 18  // (GPIO 18 บน IPST-WiFi หรือ D4/GPIO 2 บน AX-WiFi)

bool lastButtonState = HIGH;
bool currentButtonState = HIGH;
unsigned long buttonPressTime = 0;
bool longPressTriggered = false;
bool relayState = false;

unsigned long lastDebounceTime = 0;
unsigned long debounceDelay = 50;
unsigned long longPressDuration = 2000; 

void setup() {
  Serial.begin(115200);
  
  // 1. ตั้งค่า INPUT_PULLUP สำหรับสวิตช์ปุ่มกด และ OUTPUT สำหรับรีเลย์
  pinMode(BUTTON_PIN, [ ช่องที่ 1 ] );
  pinMode(RELAY_PIN, [ ช่องที่ 2 ] );
  
  digitalWrite(RELAY_PIN, LOW);
}

void loop() {
  int reading = digitalRead(BUTTON_PIN);
  
  if (reading != lastButtonState) {
    lastDebounceTime = millis();
  }
  
  if ((millis() - lastDebounceTime) > debounceDelay) {
    if (reading != currentButtonState) {
      currentButtonState = reading;
      
      // ตรวจจับเมื่อมีอีเวนต์การ "กดปุ่ม" (Active LOW เมื่อต่อ Pull-up)
      if (currentButtonState == [ ช่องที่ 3 ]) {
        buttonPressTime = millis();
        longPressTriggered = false;
        
        // 2. การทดลองสลับสถานะควบคุมรีเลย์ (Toggle Switch)
        relayState = [ ช่องที่ 4 ];
        digitalWrite(RELAY_PIN, relayState);
        Serial.printf("Toggle Relay: %s\n", relayState ? "ON" : "OFF");
      }
    }
  }
  
  // 3. ตรวจจับการกดค้าง (Press Long Switch) มากกว่า 2 วินาที
  if (currentButtonState == LOW && !longPressTriggered) {
    if ((millis() - buttonPressTime) > longPressDuration) {
      longPressTriggered = true;
      
      // เมื่อกดค้างสั่งดับการทำงานรีเลย์
      relayState = false;
      digitalWrite(RELAY_PIN, [ ช่องที่ 5 ]);
      Serial.println("System Reset via Long Press!");
    }
  }
  
  lastButtonState = reading;
}

ทฤษฎีที่เกี่ยวข้อง: การควบคุมเอาต์พุตด้วยอินพุตสวิตช์ปุ่มกดและ Software Debounce

1.1 วงจร Internal Pull-up Resistor

เมื่อเราต่อสวิตช์ปุ่มกดเข้ากับขาไมโครคอนโทรลเลอร์ หากไม่ต่อตัวต้านทานดึงแรงดัน ขาพินจะอยู่ในสภาวะ สัญญาณลอย (Floating State) ซึ่งไวต่อคลื่นแม่เหล็กไฟฟ้ารบกวนทำให้อ่านค่าสับสนระหว่าง 0 กับ 1

การใช้คำสั่ง pinMode(pin, INPUT_PULLUP); จะเปิดวงจรตัวต้านทานภายในดึงขาเข้าสู่ไฟ 3.3V:

  • ยังไม่กดปุ่ม: อ่านค่าได้ลอจิก HIGH (1)
  • กดปุ่ม (ต่อลงกราวด์ GND): อ่านค่าได้ลอจิก LOW (0) (Active LOW Logic)

2.1 ปรากฏการณ์ Contact Bouncing

หน้าสัมผัสโลหะของสวิตช์เชิงกล เมื่อถูกกดหรือปล่อย แผ่นโลหะจะสปริงตัวเด้งสั่นไปมาหลายครั้งในช่วงเวลา 5–20 มิลลิวินาที ส่งผลให้ไมโครคอนโทรลเลอร์ที่มีความเร็วสูงอ่านค่าผิดพลาดว่ามีการกดปุ่มซ้ำๆ หลายครั้งในการกดเพียงครั้งเดียว

3.1 การแก้ปัญหาการสั่นด้วยฟังก์ชัน millis()

การใช้ millis() หน่วงเวลาตรวจสอบสัญญาณให้นิ่ง (เช่น 50ms) โดยไม่ใช้ delay() ทำให้ CPU ไม่ถูกบล็อกการทำงาน:

int reading = digitalRead(BUTTON_PIN);
if (reading != lastButtonState) {
  lastDebounceTime = millis(); // บันทึกเวลาที่พบการเปลี่ยนสถานะ
}
if ((millis() - lastDebounceTime) > 50) {
  if (reading != currentButtonState) {
    currentButtonState = reading;
    if (currentButtonState == LOW) { /* ตรวจพบการกดปุ่มสมบูรณ์ */ }
  }
}

4.1 การตรวจจับการกดปุ่มค้าง (Long Press Detection)

บันทึกเวลา buttonPressTime = millis(); ทันทีที่เริ่มกดปุ่ม (สถานะเปลี่ยนเป็น LOW) จากนั้นในลูปตรวจสอบว่าหากปุ่มยังถูกกดค้างอยู่และ millis() - buttonPressTime > 2000 ให้สั่งรีเซ็ตระบบหรือเข้าสู่โหมดพิเศษทันที
